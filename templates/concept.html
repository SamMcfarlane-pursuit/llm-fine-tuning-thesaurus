Basic LLM Fine-Tuning Example

This example demonstrates how to fine-tune a GPT-2 model for a thesaurus application using the Hugging Face Transformers library.

LLM Fine-Tuning with Transformers 
from transformers import AutoModelForCausalLM, AutoTokenizer, TrainingArguments, Trainer
from datasets import Dataset

# Load model and tokenizer
model_name = "gpt2"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name)

# Prepare dataset
data = [
    "A synonym for happy is joyful.",
    "A synonym for sad is unhappy.",
    "A synonym for big is large."
]
dataset = Dataset.from_dict({"text": data})
dataset = dataset.train_test_split(test_size=0.2)

# Tokenize dataset
def tokenize_function(examples):
    return tokenizer(examples["text"], padding="max_length", truncation=True, max_length=128)

tokenized_datasets = dataset.map(tokenize_function, batched=True)

# Set up training arguments
training_args = TrainingArguments(
    output_dir="./results",
    evaluation_strategy="epoch",
    learning_rate=2e-5,
    per_device_train_batch_size=8,
    per_device_eval_batch_size=8,
    num_train_epochs=3,
    weight_decay=0.01,
)

# Create trainer
trainer = Trainer(
    model=model,
    args=training_args,
    train_dataset=tokenized_datasets["train"],
    eval_dataset=tokenized_datasets["test"],
)

# Fine-tune the model
trainer.train()
Pro Tip: For larger models, consider using parameter-efficient fine-tuning methods like LoRA or QLoRA to reduce memory requirements.

LoRA Implementation Example

This example shows how to apply LoRA to a pre-trained GPT-2 model using the PEFT library, significantly reducing the number of trainable parameters.

LoRA Implementation with PEFT 
from transformers import AutoModelForCausalLM, AutoTokenizer
from peft import LoraConfig, get_peft_model, TaskType

# Load model and tokenizer
model_name = "gpt2"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(model_name)

# Configure LoRA
lora_config = LoraConfig(
    task_type=TaskType.CAUSAL_LM,
    r=8,                     # Rank of the update matrices
    lora_alpha=16,           # Parameter for scaling (typically 2x rank)
    lora_dropout=0.1,        # Dropout probability for LoRA layers
    bias="none",             # Don't add bias parameters
    target_modules=["c_attn", "c_proj"]  # Attention modules to apply LoRA to
)

# Apply LoRA to the model
peft_model = get_peft_model(model, lora_config)

# Print trainable parameters
trainable_params = sum(p.numel() for p in peft_model.parameters() if p.requires_grad)
total_params = sum(p.numel() for p in peft_model.parameters())
print(f"Trainable parameters: {trainable_params:,} ({trainable_params/total_params:.2%} of total)")
Pro Tip: The rank parameter (r) controls the trade-off between model capacity and efficiency. Start with smaller values (4-8) and increase if needed.
{% elif concept.name == "QLoRA" %}

QLoRA Implementation Example

This example demonstrates how to implement QLoRA, combining 4-bit quantization with LoRA for extreme memory efficiency.

QLoRA Implementation 
from transformers import AutoModelForCausalLM, AutoTokenizer, BitsAndBytesConfig
from peft import LoraConfig, get_peft_model, TaskType
import torch

# Configure 4-bit quantization
bnb_config = BitsAndBytesConfig(
    load_in_4bit=True,
    bnb_4bit_use_double_quant=True,  # Use double quantization
    bnb_4bit_quant_type="nf4",       # Normalized float 4
    bnb_4bit_compute_dtype=torch.float16
)

# Load model with quantization
model_name = "meta-llama/Llama-2-7b-hf"  # Example larger model
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    quantization_config=bnb_config,
    device_map="auto"
)
tokenizer = AutoTokenizer.from_pretrained(model_name)

# Configure LoRA for the quantized model
lora_config = LoraConfig(
    r=8,
    lora_alpha=16,
    target_modules=["q_proj", "k_proj", "v_proj", "o_proj"],
    lora_dropout=0.05,
    bias="none",
    task_type=TaskType.CAUSAL_LM
)

# Apply LoRA to the quantized model
qlora_model = get_peft_model(model, lora_config)

# Print trainable parameters
trainable_params = sum(p.numel() for p in qlora_model.parameters() if p.requires_grad)
print(f"Trainable parameters: {trainable_params:,}")
Pro Tip: QLoRA is particularly useful for fine-tuning larger models (7B+ parameters) on consumer hardware with limited VRAM.

Thesaurus Training Data Preparation Example

This example shows how to prepare training data for a thesaurus application using WordNet as a data source.

Training Data Preparation 
import nltk
from nltk.corpus import wordnet as wn
import json

# Download WordNet
nltk.download('wordnet')

# Generate thesaurus data
words = ["happy", "good", "fast", "big", "small"]
thesaurus_data = []

for word in words:
    entry = {"word": word, "synonyms": [], "antonyms": []}

    # Get synsets for the word
    synsets = wn.synsets(word)

    for synset in synsets:
        # Get synonyms
        for lemma in synset.lemmas():
            if lemma.name() != word and '_' not in lemma.name():
                entry["synonyms"].append(lemma.name())

            # Get antonyms
            for antonym in lemma.antonyms():
                if '_' not in antonym.name():
                    entry["antonyms"].append(antonym.name())

    # Remove duplicates
    entry["synonyms"] = list(set(entry["synonyms"]))
    entry["antonyms"] = list(set(entry["antonyms"]))

    thesaurus_data.append(entry)

# Convert to instruction-following format for better fine-tuning
training_examples = []
for entry in thesaurus_data:
    word = entry["word"]
    if entry["synonyms"]:
        synonyms_str = ", ".join(entry["synonyms"])
        training_examples.append({
            "instruction": f"List synonyms for the word '{word}'.",
            "response": f"Synonyms for '{word}': {synonyms_str}"
        })

print(f"Generated {len(training_examples)} training examples")
Pro Tip: Using an instruction-following format helps the model learn to respond to user queries more effectively.
