Step 2: Custom Image Dataset Class

Python 
class CustomImageDataset(Dataset):
    def __init__(self, image_paths, labels, class_names, transform=None, target_transform=None):
        """
        Custom dataset for image classification
        
        Args:
            image_paths: List of paths to images
            labels: List of corresponding labels
            class_names: List of class names
            transform: Optional transform to be applied to images
            target_transform: Optional transform to be applied to labels
        """
        self.image_paths = image_paths
        self.labels = labels
        self.class_names = class_names
        self.transform = transform
        self.target_transform = target_transform
        
        # Validate data
        assert len(image_paths) == len(labels), "Number of images and labels must match"
        
        print(f"Dataset initialized with {len(self)} samples")
        print(f"Classes: {class_names}")
    
    def __len__(self):
        return len(self.image_paths)
    
    def __getitem__(self, idx):
        # Get image path and label
        image_path = self.image_paths[idx]
        label = self.labels[idx]
        
        # For demonstration, create a random image if file doesn't exist
        try:
            image = Image.open(image_path).convert('RGB')
        except (FileNotFoundError, OSError):
            # Create a random image for demonstration
            image = Image.fromarray(np.random.randint(0, 255, (224, 224, 3), dtype=np.uint8))
        
        # Apply transforms
        if self.transform:
            image = self.transform(image)
        
        if self.target_transform:
            label = self.target_transform(label)
        
        return image, label
    
    def get_class_name(self, label):
        return self.class_names[label]
    
    def get_class_distribution(self):
        """Get the distribution of classes in the dataset"""
        unique, counts = np.unique(self.labels, return_counts=True)
        distribution = dict(zip([self.class_names[i] for i in unique], counts))
        return distribution

Step 4: Visualization and Analysis

Python 
def denormalize_image(tensor, mean, std):
    """Denormalize image tensor for visualization"""
    for t, m, s in zip(tensor, mean, std):
        t.mul_(s).add_(m)
    return torch.clamp(tensor, 0, 1)

def visualize_batch(data_loader, dataset, num_samples=8):
    """Visualize a batch of images with their labels"""
    # Get a batch
    images, labels = next(iter(data_loader))
    
    # Denormalize images for visualization
    mean = [0.485, 0.456, 0.406]
    std = [0.229, 0.224, 0.225]
    
    fig, axes = plt.subplots(2, 4, figsize=(12, 6))
    axes = axes.ravel()
    
    for i in range(min(num_samples, len(images))):
        # Denormalize image
        img = denormalize_image(images[i].clone(), mean, std)
        img = img.permute(1, 2, 0).numpy()
        
        # Plot image
        axes[i].imshow(img)
        axes[i].set_title(f'{dataset.get_class_name(labels[i].item())}')
        axes[i].axis('off')
    
    plt.tight_layout()
    plt.show()

# Visualize training data with augmentations
print("Training data with augmentations:")
visualize_batch(train_loader, train_dataset)

# Visualize validation data (no augmentations)
print("Validation data (no augmentations):")
visualize_batch(val_loader, val_dataset)

Exercise 2: Text Dataset with Variable Lengths

Create a custom dataset for text data with proper handling of variable-length sequences.

Text Dataset Implementation

Python 
from torch.nn.utils.rnn import pad_sequence
import re
from collections import Counter

class TextDataset(Dataset):
    def __init__(self, texts, labels, vocab=None, max_length=None):
        self.texts = texts
        self.labels = labels
        self.max_length = max_length
        
        # Build vocabulary if not provided
        if vocab is None:
            self.vocab = self.build_vocabulary(texts)
        else:
            self.vocab = vocab
        
        # Convert texts to token indices
        self.encoded_texts = [self.encode_text(text) for text in texts]
        
    def build_vocabulary(self, texts):
        """Build vocabulary from texts"""
        vocab = {'': 0, '': 1, '': 2, '': 3}
        word_counts = Counter()
        
        for text in texts:
            tokens = self.tokenize(text)
            word_counts.update(tokens)
        
        # Add words to vocabulary (frequency threshold can be added)
        for word, count in word_counts.most_common():
            if word not in vocab:
                vocab[word] = len(vocab)
        
        return vocab
    
    def tokenize(self, text):
        """Simple tokenization"""
        text = re.sub(r'[^\w\s]', '', text.lower())
        return text.split()
    
    def encode_text(self, text):
        """Convert text to sequence of token indices"""
        tokens = self.tokenize(text)
        encoded = [self.vocab.get(token, 1) for token in tokens]  # 1 is 
        
        # Add start and end tokens
        encoded = [2] + encoded + [3]  #  + tokens + 
        
        # Truncate if too long
        if self.max_length and len(encoded) > self.max_length:
            encoded = encoded[:self.max_length-1] + [3]  # Keep 
        
        return encoded
    
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        return torch.tensor(self.encoded_texts[idx], dtype=torch.long), torch.tensor(self.labels[idx], dtype=torch.long)

# Sample text data

# Custom collate function for text data
def text_collate_fn(batch):
    texts, labels = zip(*batch)
    
    # Pad sequences to same length
    padded_texts = pad_sequence(texts, batch_first=True, padding_value=0)
    labels = torch.stack(labels)
    
    # Get actual lengths (before padding)
    lengths = [len(text) for text in texts]
    
    return padded_texts, labels, lengths

# Create text data loader
text_loader = DataLoader(text_dataset, batch_size=3, shuffle=True, collate_fn=text_collate_fn)
