Step 1: Create and Train a Simple Model

Python 
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader
import os
import json

# Device configuration
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f'Using device: {device}')

# Simple CNN for CIFAR-10
class SimpleCNN(nn.Module):
    def __init__(self, num_classes=10):
        super(SimpleCNN, self).__init__()
        self.features = nn.Sequential(
            nn.Conv2d(3, 32, 3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(32, 64, 3, padding=1),
            nn.ReLU(),
            nn.MaxPool2d(2),
            nn.Conv2d(64, 128, 3, padding=1),
            nn.ReLU(),
            nn.AdaptiveAvgPool2d((4, 4))
        )
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Linear(128 * 4 * 4, 256),
            nn.ReLU(),
            nn.Dropout(0.5),
            nn.Linear(256, num_classes)
        )
    
    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return x

# Create model
model = SimpleCNN(num_classes=10).to(device)
print(f"Model created with {sum(p.numel() for p in model.parameters()):,} parameters")

# Quick training setup (for demonstration)
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))
])

# Load small subset for quick training
trainset = torchvision.datasets.CIFAR10(root='./data', train=True, download=True, transform=transform)
train_loader = DataLoader(trainset, batch_size=64, shuffle=True)

criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# Quick training (just 1 epoch for demonstration)
model.train()
for batch_idx, (data, targets) in enumerate(train_loader):
    if batch_idx >= 10:  # Just train on a few batches
        break
    
    data, targets = data.to(device), targets.to(device)
    
    optimizer.zero_grad()
    outputs = model(data)
    loss = criterion(outputs, targets)
    loss.backward()
    optimizer.step()
    
    if batch_idx % 5 == 0:
        print(f'Batch {batch_idx}, Loss: {loss.item():.4f}')

print("Quick training completed!")

Step 3: Load Model for Inference

Python 
# Function to load model from checkpoint
def load_model_from_checkpoint(checkpoint_path, config_path=None):
    # Load configuration
    if config_path and os.path.exists(config_path):
        with open(config_path, 'r') as f:
            config = json.load(f)
    else:
        config = {'num_classes': 10}  # Default config
    
    # Create model instance
    model = SimpleCNN(num_classes=config['num_classes'])
    
    # Load checkpoint
    checkpoint = torch.load(checkpoint_path, map_location=device)
    
    # Load state dict
    model.load_state_dict(checkpoint['model_state_dict'])
    
    # Set to evaluation mode
    model.eval()
    
    print(f"Model loaded from {checkpoint_path}")
    print(f"Training info: {checkpoint.get('training_info', 'Not available')}")
    
    return model, config

# Load the model
loaded_model, config = load_model_from_checkpoint(
    'models/model_checkpoint.pth', 
    'models/model_config.json'
)

print(f"Loaded model configuration: {config}")

# Test inference
test_input = torch.randn(1, 3, 32, 32).to(device)
with torch.no_grad():
    output = loaded_model(test_input)
    predicted_class = torch.argmax(output, dim=1).item()
    confidence = torch.softmax(output, dim=1).max().item()

print(f"Test inference - Predicted class: {predicted_class}, Confidence: {confidence:.3f}")

Step 4: TorchScript Conversion

Python 
# Convert model to TorchScript for production deployment
model.eval()

# Method 1: Tracing (recommended for most cases)
example_input = torch.randn(1, 3, 32, 32).to(device)
traced_model = torch.jit.trace(model, example_input)

# Save traced model
traced_model.save('models/traced_model.pt')
print("Traced model saved")

# Method 2: Scripting (for models with control flow)
try:
    scripted_model = torch.jit.script(model)
    scripted_model.save('models/scripted_model.pt')
    print("Scripted model saved")
except Exception as e:
    print(f"Scripting failed: {e}")

# Load and test TorchScript model
loaded_traced_model = torch.jit.load('models/traced_model.pt')
loaded_traced_model.eval()

# Test TorchScript model
with torch.no_grad():
    original_output = model(example_input)
    traced_output = loaded_traced_model(example_input)
    
    # Check if outputs are similar
    diff = torch.abs(original_output - traced_output).max().item()
    print(f"Max difference between original and traced model: {diff:.6f}")

# Benchmark inference speed
import time

def benchmark_model(model, input_tensor, num_runs=100):
    model.eval()
    with torch.no_grad():
        # Warmup
        for _ in range(10):
            _ = model(input_tensor)
        
        # Benchmark
        start_time = time.time()
        for _ in range(num_runs):
            _ = model(input_tensor)
        end_time = time.time()
    
    avg_time = (end_time - start_time) / num_runs * 1000  # ms
    return avg_time

original_time = benchmark_model(model, example_input)
traced_time = benchmark_model(loaded_traced_model, example_input)

print(f"Original model inference time: {original_time:.2f} ms")
print(f"Traced model inference time: {traced_time:.2f} ms")
print(f"Speedup: {original_time / traced_time:.2f}x")

Step 5: Simple REST API for Model Serving

Python 
# Simple Flask API for model serving
from flask import Flask, request, jsonify
import base64
from PIL import Image
import io
import numpy as np

app = Flask(__name__)

# Load model once at startup
MODEL_PATH = 'models/traced_model.pt'
CONFIG_PATH = 'models/model_config.json'

# Load configuration
with open(CONFIG_PATH, 'r') as f:
    model_config = json.load(f)

# Load TorchScript model
model = torch.jit.load(MODEL_PATH)
model.eval()

print("Model loaded for serving")

# Preprocessing function
def preprocess_image(image_data):
    """Preprocess image for model inference"""
    # Decode base64 image
    image_bytes = base64.b64decode(image_data)
    image = Image.open(io.BytesIO(image_bytes)).convert('RGB')
    
    # Resize to model input size
    image = image.resize((32, 32))
    
    # Convert to tensor and normalize
    transform = transforms.Compose([
        transforms.ToTensor(),
        transforms.Normalize((0.5, 0.5, 0.5), (0.5, 0.5, 0.5))
    ])
    
    tensor = transform(image).unsqueeze(0)  # Add batch dimension
    return tensor

@app.route('/predict', methods=['POST'])
def predict():
    try:
        # Get image data from request
        data = request.get_json()
        image_data = data['image']  # Base64 encoded image
        
        # Preprocess image
        input_tensor = preprocess_image(image_data)
        
        # Run inference
        with torch.no_grad():
            output = model(input_tensor)
            probabilities = torch.softmax(output, dim=1)
            predicted_class = torch.argmax(output, dim=1).item()
            confidence = probabilities.max().item()
        
        # Prepare response
        response = {
            'predicted_class': predicted_class,
            'class_name': model_config['class_names'][predicted_class],
            'confidence': float(confidence),
            'all_probabilities': probabilities.squeeze().tolist()
        }
        
        return jsonify(response)
    
    except Exception as e:
        return jsonify({'error': str(e)}), 400

@app.route('/health', methods=['GET'])
def health():
    return jsonify({'status': 'healthy', 'model': 'SimpleCNN'})

@app.route('/info', methods=['GET'])
def info():
    return jsonify(model_config)

# Example of how to run the server
if __name__ == '__main__':
    print("Starting model serving API...")
    print("Endpoints:")
    print("  POST /predict - Make predictions")
    print("  GET /health - Health check")
    print("  GET /info - Model information")
    
    # Note: In production, use a proper WSGI server like Gunicorn
    # app.run(host='0.0.0.0', port=5000, debug=False)
    
    print("API code ready (uncomment app.run() to start server)")

# Example client code
def example_client():
    """Example of how to call the API"""
    import requests
    
    # Create a dummy image (in practice, you'd encode a real image)
    dummy_image = Image.new('RGB', (32, 32), color='red')
    buffer = io.BytesIO()
    dummy_image.save(buffer, format='PNG')
    image_base64 = base64.b64encode(buffer.getvalue()).decode()
    
    # Make prediction request
    response = requests.post('http://localhost:5000/predict', 
                           json={'image': image_base64})
    
    if response.status_code == 200:
        result = response.json()
        print(f"Prediction: {result['class_name']} (confidence: {result['confidence']:.3f})")
    else:
        print(f"Error: {response.text}")

print("Example client function created")
