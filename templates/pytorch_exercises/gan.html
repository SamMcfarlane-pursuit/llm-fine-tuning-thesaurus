Exercise 1: Simple GAN for MNIST

Build a basic GAN to generate handwritten digits using the MNIST dataset.

Step 1: Generator Network

Python 
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader
import matplotlib.pyplot as plt
import numpy as np

# Device configuration
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f'Using device: {device}')

class Generator(nn.Module):
    def __init__(self, noise_dim=100, img_dim=784):
        super(Generator, self).__init__()
        self.noise_dim = noise_dim
        self.img_dim = img_dim
        
        self.model = nn.Sequential(
            # First layer
            nn.Linear(noise_dim, 256),
            nn.LeakyReLU(0.2),
            nn.BatchNorm1d(256),
            
            # Second layer
            nn.Linear(256, 512),
            nn.LeakyReLU(0.2),
            nn.BatchNorm1d(512),
            
            # Third layer
            nn.Linear(512, 1024),
            nn.LeakyReLU(0.2),
            nn.BatchNorm1d(1024),
            
            # Output layer
            nn.Linear(1024, img_dim),
            nn.Tanh()  # Output between -1 and 1
        )
    
    def forward(self, noise):
        return self.model(noise)

# Test generator
noise_dim = 100
img_dim = 28 * 28  # MNIST image size
generator = Generator(noise_dim, img_dim).to(device)

# Generate sample
sample_noise = torch.randn(4, noise_dim).to(device)
fake_images = generator(sample_noise)

print(f"Generator architecture:\n{generator}")
print(f"Input noise shape: {sample_noise.shape}")
print(f"Generated images shape: {fake_images.shape}")
print(f"Generator parameters: {sum(p.numel() for p in generator.parameters()):,}")

Step 2: Discriminator Network

Python 
class Discriminator(nn.Module):
    def __init__(self, img_dim=784):
        super(Discriminator, self).__init__()
        self.img_dim = img_dim
        
        self.model = nn.Sequential(
            # First layer
            nn.Linear(img_dim, 1024),
            nn.LeakyReLU(0.2),
            nn.Dropout(0.3),
            
            # Second layer
            nn.Linear(1024, 512),
            nn.LeakyReLU(0.2),
            nn.Dropout(0.3),
            
            # Third layer
            nn.Linear(512, 256),
            nn.LeakyReLU(0.2),
            nn.Dropout(0.3),
            
            # Output layer
            nn.Linear(256, 1),
            nn.Sigmoid()  # Output probability
        )
    
    def forward(self, img):
        return self.model(img)

# Test discriminator
discriminator = Discriminator(img_dim).to(device)

# Test with real and fake images
real_images = torch.randn(4, img_dim).to(device)  # Placeholder for real images
fake_images = generator(sample_noise)

real_output = discriminator(real_images)
fake_output = discriminator(fake_images.detach())

print(f"Discriminator architecture:\n{discriminator}")
print(f"Real images output: {real_output.squeeze()}")
print(f"Fake images output: {fake_output.squeeze()}")
print(f"Discriminator parameters: {sum(p.numel() for p in discriminator.parameters()):,}")

Step 3: Data Preparation

Python 
# Data preprocessing for GAN
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.5,), (0.5,))  # Normalize to [-1, 1]
])

# Load MNIST dataset
dataset = torchvision.datasets.MNIST(
    root='./data', 
    train=True, 
    transform=transform, 
    download=True
)

# Create data loader
batch_size = 64
data_loader = DataLoader(dataset, batch_size=batch_size, shuffle=True)

print(f"Dataset size: {len(dataset)}")
print(f"Number of batches: {len(data_loader)}")

# Visualize some real images
def show_images(images, title="Images", nrow=8):
    # Denormalize images from [-1, 1] to [0, 1]
    images = (images + 1) / 2
    
    # Create grid
    grid = torchvision.utils.make_grid(images, nrow=nrow, padding=2)
    
    # Plot
    plt.figure(figsize=(10, 10))
    plt.imshow(grid.permute(1, 2, 0))
    plt.title(title)
    plt.axis('off')
    plt.show()

# Show sample real images
sample_batch, _ = next(iter(data_loader))
show_images(sample_batch[:16], "Real MNIST Images")

Step 4: GAN Training Loop

Python 
# Training setup
lr = 0.0002
beta1 = 0.5  # Adam optimizer parameter for GANs

# Optimizers
optimizer_G = optim.Adam(generator.parameters(), lr=lr, betas=(beta1, 0.999))
optimizer_D = optim.Adam(discriminator.parameters(), lr=lr, betas=(beta1, 0.999))

# Loss function
criterion = nn.BCELoss()

# Training function
def train_gan(generator, discriminator, data_loader, num_epochs=50):
    # Lists to track losses
    G_losses = []
    D_losses = []
    
    # Fixed noise for visualization
    fixed_noise = torch.randn(16, noise_dim).to(device)
    
    print("Starting GAN training...")
    
    for epoch in range(num_epochs):
        for batch_idx, (real_images, _) in enumerate(data_loader):
            batch_size = real_images.size(0)
            real_images = real_images.view(batch_size, -1).to(device)
            
            # Labels
            real_labels = torch.ones(batch_size, 1).to(device)
            fake_labels = torch.zeros(batch_size, 1).to(device)
            
            # ===============================
            # Train Discriminator
            # ===============================
            optimizer_D.zero_grad()
            
            # Real images
            real_output = discriminator(real_images)
            d_loss_real = criterion(real_output, real_labels)
            
            # Fake images
            noise = torch.randn(batch_size, noise_dim).to(device)
            fake_images = generator(noise)
            fake_output = discriminator(fake_images.detach())
            d_loss_fake = criterion(fake_output, fake_labels)
            
            # Total discriminator loss
            d_loss = d_loss_real + d_loss_fake
            d_loss.backward()
            optimizer_D.step()
            
            # ===============================
            # Train Generator
            # ===============================
            optimizer_G.zero_grad()
            
            # Generate fake images and get discriminator output
            fake_output = discriminator(fake_images)
            g_loss = criterion(fake_output, real_labels)  # Want discriminator to think fake is real
            
            g_loss.backward()
            optimizer_G.step()
            
            # Track losses
            G_losses.append(g_loss.item())
            D_losses.append(d_loss.item())
            
            # Print progress
            if batch_idx % 200 == 0:
                print(f'Epoch [{epoch+1}/{num_epochs}], Step [{batch_idx}/{len(data_loader)}], '
                      f'D_loss: {d_loss.item():.4f}, G_loss: {g_loss.item():.4f}')
        
        # Generate images for visualization
        if (epoch + 1) % 10 == 0:
            with torch.no_grad():
                fake_images = generator(fixed_noise)
                fake_images = fake_images.view(-1, 1, 28, 28)
                show_images(fake_images, f"Generated Images - Epoch {epoch+1}")
    
    return G_losses, D_losses

# Train the GAN
print("Training GAN...")
G_losses, D_losses = train_gan(generator, discriminator, data_loader, num_epochs=20)
print("Training completed!")
