Exercise 1: Feature Extraction with Pre-trained ResNet

Use a pre-trained ResNet model as a feature extractor for a custom classification task.

Step 1: Load Pre-trained Model

Python 
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
import torchvision.models as models
from torch.utils.data import DataLoader, Dataset
import matplotlib.pyplot as plt
import numpy as np

# Device configuration
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
print(f'Using device: {device}')

# Load pre-trained ResNet18
model = models.resnet18(pretrained=True)
print("Pre-trained ResNet18 loaded successfully!")

# Print model architecture
print(f"Model architecture:\n{model}")

# Check the final layer
print(f"Original final layer: {model.fc}")
print(f"Number of features in final layer: {model.fc.in_features}")

# Freeze all parameters for feature extraction
for param in model.parameters():
    param.requires_grad = False

print("All parameters frozen for feature extraction")

Step 2: Modify Model for Custom Task

Python 
# Custom classifier for our task
class TransferLearningModel(nn.Module):
    def __init__(self, pretrained_model, num_classes, feature_extract=True):
        super(TransferLearningModel, self).__init__()
        self.feature_extract = feature_extract
        
        # Use pre-trained model as feature extractor
        self.features = nn.Sequential(*list(pretrained_model.children())[:-1])  # Remove final FC layer
        
        # Add custom classifier
        self.classifier = nn.Sequential(
            nn.Flatten(),
            nn.Dropout(0.5),
            nn.Linear(pretrained_model.fc.in_features, 256),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(256, num_classes)
        )
        
        # Freeze feature extractor if specified
        if feature_extract:
            for param in self.features.parameters():
                param.requires_grad = False
    
    def forward(self, x):
        x = self.features(x)
        x = self.classifier(x)
        return x

# Create model for 3-class classification (example: cats, dogs, birds)
num_classes = 3
transfer_model = TransferLearningModel(model, num_classes, feature_extract=True).to(device)

print(f"Transfer learning model created for {num_classes} classes")
print(f"Model: {transfer_model}")

# Count trainable parameters
trainable_params = sum(p.numel() for p in transfer_model.parameters() if p.requires_grad)
total_params = sum(p.numel() for p in transfer_model.parameters())
print(f"Trainable parameters: {trainable_params:,}")
print(f"Total parameters: {total_params:,}")
print(f"Percentage trainable: {100 * trainable_params / total_params:.2f}%")

Step 3: Data Preparation with Proper Transforms

Python 
# Data transforms for transfer learning
# Important: Use ImageNet normalization for pre-trained models
data_transforms = {
    'train': transforms.Compose([
        transforms.Resize(256),
        transforms.RandomCrop(224),
        transforms.RandomHorizontalFlip(),
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])  # ImageNet stats
    ]),
    'val': transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize([0.485, 0.456, 0.406], [0.229, 0.224, 0.225])
    ])
}

# Create sample dataset (using CIFAR-10 as example)
# In practice, you would use your own dataset
transform_train = data_transforms['train']
transform_val = data_transforms['val']

# Load CIFAR-10 and use only 3 classes for demonstration
full_trainset = torchvision.datasets.CIFAR10(root='./data', train=True, download=True, transform=transform_train)
full_testset = torchvision.datasets.CIFAR10(root='./data', train=False, download=True, transform=transform_val)

# CIFAR-10 classes: 0=airplane, 1=automobile, 2=bird, 3=cat, 4=deer, 5=dog, 6=frog, 7=horse, 8=ship, 9=truck
# Let's use classes 2 (bird), 3 (cat), 5 (dog) and map them to 0, 1, 2
selected_classes = [2, 3, 5]  # bird, cat, dog
class_mapping = {2: 0, 3: 1, 5: 2}  # map to 0, 1, 2

def filter_dataset(dataset, selected_classes, class_mapping):
    """Filter dataset to include only selected classes"""
    filtered_data = []
    filtered_targets = []
    
    for i, (data, target) in enumerate(dataset):
        if target in selected_classes:
            filtered_data.append(data)
            filtered_targets.append(class_mapping[target])
    
    return filtered_data, filtered_targets

# Filter datasets
train_data, train_targets = filter_dataset(full_trainset, selected_classes, class_mapping)
test_data, test_targets = filter_dataset(full_testset, selected_classes, class_mapping)

print(f"Filtered train dataset: {len(train_data)} samples")
print(f"Filtered test dataset: {len(test_data)} samples")

# Create custom dataset class
class FilteredDataset(Dataset):
    def __init__(self, data, targets):
        self.data = data
        self.targets = targets
    
    def __len__(self):
        return len(self.data)
    
    def __getitem__(self, idx):
        return self.data[idx], self.targets[idx]

# Create datasets and data loaders
train_dataset = FilteredDataset(train_data, train_targets)
test_dataset = FilteredDataset(test_data, test_targets)

batch_size = 32
train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)
test_loader = DataLoader(test_dataset, batch_size=batch_size, shuffle=False)

print(f"Train loader: {len(train_loader)} batches")
print(f"Test loader: {len(test_loader)} batches")

Step 4: Training with Transfer Learning

Python 
# Training setup
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(transfer_model.parameters(), lr=0.001)
scheduler = optim.lr_scheduler.StepLR(optimizer, step_size=7, gamma=0.1)

# Training function
def train_transfer_model(model, train_loader, test_loader, criterion, optimizer, scheduler, num_epochs=10):
    train_losses = []
    train_accuracies = []
    test_accuracies = []
    
    for epoch in range(num_epochs):
        # Training phase
        model.train()
        running_loss = 0.0
        correct = 0
        total = 0
        
        for batch_idx, (data, targets) in enumerate(train_loader):
            data, targets = data.to(device), targets.to(device)
            
            # Forward pass
            outputs = model(data)
            loss = criterion(outputs, targets)
            
            # Backward pass
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            # Statistics
            running_loss += loss.item()
            _, predicted = torch.max(outputs.data, 1)
            total += targets.size(0)
            correct += (predicted == targets).sum().item()
            
            if batch_idx % 50 == 0:
                print(f'Epoch [{epoch+1}/{num_epochs}], Step [{batch_idx}/{len(train_loader)}], '
                      f'Loss: {loss.item():.4f}')
        
        # Calculate training accuracy
        train_acc = 100 * correct / total
        epoch_loss = running_loss / len(train_loader)
        train_losses.append(epoch_loss)
        train_accuracies.append(train_acc)
        
        # Evaluation phase
        model.eval()
        test_correct = 0
        test_total = 0
        
        with torch.no_grad():
            for data, targets in test_loader:
                data, targets = data.to(device), targets.to(device)
                outputs = model(data)
                _, predicted = torch.max(outputs, 1)
                test_total += targets.size(0)
                test_correct += (predicted == targets).sum().item()
        
        test_acc = 100 * test_correct / test_total
        test_accuracies.append(test_acc)
        
        # Update learning rate
        scheduler.step()
        
        print(f'Epoch [{epoch+1}/{num_epochs}] - Train Loss: {epoch_loss:.4f}, '
              f'Train Acc: {train_acc:.2f}%, Test Acc: {test_acc:.2f}%')
    
    return train_losses, train_accuracies, test_accuracies

# Train the model
print("Starting transfer learning training...")
train_losses, train_accs, test_accs = train_transfer_model(
    transfer_model, train_loader, test_loader, criterion, optimizer, scheduler, num_epochs=10
)
print("Training completed!")

Step 5: Compare with Fine-tuning

Python 
# Create fine-tuning model (unfreeze some layers)
finetune_model = TransferLearningModel(models.resnet18(pretrained=True), num_classes, feature_extract=False).to(device)

# Freeze early layers, unfreeze later layers
for name, param in finetune_model.named_parameters():
    if 'features.0' in name or 'features.1' in name or 'features.2' in name or 'features.3' in name:
        param.requires_grad = False  # Freeze early layers
    else:
        param.requires_grad = True   # Unfreeze later layers

# Count trainable parameters for fine-tuning
finetune_trainable = sum(p.numel() for p in finetune_model.parameters() if p.requires_grad)
finetune_total = sum(p.numel() for p in finetune_model.parameters())
print(f"Fine-tuning - Trainable parameters: {finetune_trainable:,}")
print(f"Fine-tuning - Percentage trainable: {100 * finetune_trainable / finetune_total:.2f}%")

# Use lower learning rate for fine-tuning
finetune_optimizer = optim.Adam(finetune_model.parameters(), lr=0.0001)
finetune_scheduler = optim.lr_scheduler.StepLR(finetune_optimizer, step_size=7, gamma=0.1)

# Train fine-tuning model (shorter training for comparison)
print("\nStarting fine-tuning training...")
finetune_losses, finetune_train_accs, finetune_test_accs = train_transfer_model(
    finetune_model, train_loader, test_loader, criterion, finetune_optimizer, finetune_scheduler, num_epochs=5
)

# Plot comparison
plt.figure(figsize=(15, 5))
